async function updateRadar() { // 1. REPAIRED AVIATION ENDPOINT PATH const targetUrl = 'https://adsb.fi' + homeLat + '/' + homeLon + '/150'; // 2. REPAIRED PROXY HOST AND ROUTING PARAMETERS const url = 'https://allorigins.win' + encodeURIComponent(targetUrl); try { const response = await fetch(url); if (!response.ok) throw new Error("Server communication issue"); const rawData = await response.json(); // Safely unpack the contents text string from the proxy server let data; if (typeof rawData.contents === 'string') { data = JSON.parse(rawData.contents); } else { data = rawData.contents; } const count = data && data.ac ? data.ac.length : 0; document.getElementById('target-count').innerText = `TRACKING: ${count} TARGETS`; let activeTargets = {}; if (data && data.ac) { data.ac.forEach(aircraft => { const callsign = (aircraft.flight || aircraft.registration || "UNK").trim(); const lat = aircraft.lat; const lon = aircraft.lon; const alt = aircraft.alt_baro ? `${aircraft.alt_baro} FT` : "GND"; const spd = aircraft.gs ? `${Math.round(aircraft.gs)} KTS` : "---"; if (lat && lon) { activeTargets[callsign] = true; const hudPopup = `ID: ${callsign}
ALT: ${alt}
SPD: ${spd}`; if (aircraftMarkers[callsign]) { aircraftMarkers[callsign].setLatLng([lat, lon]); aircraftMarkers[callsign].getPopup().setContent(hudPopup); } else { // Draw bright tactical radar vector blip aircraftMarkers[callsign] = L.circleMarker([lat, lon], { color: '#00ff66', fillColor: '#0b0e14', fillOpacity: 0.9, weight: 2, radius: 6 }).addTo(map).bindPopup(hudPopup); // Attach permanent moving text label next to the blip aircraftMarkers[callsign].bindTooltip(callsign, { permanent: true, direction: 'right', className: 'radar-label', offset: [5, 0] }); } } }); } // Automatically clear aircraft leaving our 150km grid for (let callsign in aircraftMarkers) { if (!activeTargets[callsign]) { map.removeLayer(aircraftMarkers[callsign]); delete aircraftMarkers[callsign]; } } } catch (error) { console.error("Radar Sync Interrupted:", error); document.getElementById('target-count').innerText = "RADAR SYNC ERROR"; } }
